 * All original features intact. Speed-only changes vs v3:
 *
 *  - Prefers gemini-2.5-flash-lite. If Google 404s it for this API key (common for

 *    then gemini-3.1-flash-lite. Still ONE model per run after the first hit.

 *  - Batch path sends the PDF on the first round (skips slow Apps Script text scraping
 *    that was causing a wasted Gemini call + 2.5s pause + full-PDF retry).
 *  - Retry only for transient errors (503 / 429 / empty / parse), same model, no 2.5s sleep.
 *  - PDF base64 is encoded once and reused if a retry is needed.
 * ======================================================================================
 */

/**
 * Real-time reactive trigger: regenerates WhatsApp & Gmail links on edit.
 */
function onEdit(e) {
  if (!e || !e.range) return;
  try {
    const sheet = e.range.getSheet();
    if (sheet.getName() !== CONFIG.SHEET_MASTER_DATA) return;

    const row = e.range.getRow();
    const col = e.range.getColumn();
    if (row <= 1) return;

    if (
      col === CONFIG.COLS.CLIENT_MOBILE ||
      col === CONFIG.COLS.CLIENT_EMAIL ||
      col === CONFIG.COLS.CLIENT_NAME ||
      col === CONFIG.COLS.RENEWAL_DATE ||
      col === CONFIG.COLS.COMPANY ||
      col === CONFIG.COLS.POLICY_NO
    ) {
      const rowData = sheet.getRange(row, 1, 1, CONFIG.COLS.NOTES).getValues()[0];
      const clientName = String(rowData[CONFIG.COLS.CLIENT_NAME - 1] || '');
      const renewalDate = String(rowData[CONFIG.COLS.RENEWAL_DATE - 1] || '');
      const company = String(rowData[CONFIG.COLS.COMPANY - 1] || '');
      const policyNo = String(rowData[CONFIG.COLS.POLICY_NO - 1] || '');
      const mobile = String(rowData[CONFIG.COLS.CLIENT_MOBILE - 1] || '');
      const email = String(rowData[CONFIG.COLS.CLIENT_EMAIL - 1] || '');
      const rawFileId = String(rowData[CONFIG.COLS.FILE_ID - 1] || '');

      let fileFormula = '';
      try { fileFormula = sheet.getRange(row, CONFIG.COLS.FILE_LINK).getFormula(); } catch (e2) {}

      const fileId = extractDriveFileId(rawFileId) || extractDriveFileId(fileFormula);
      const fileUrl = buildDriveViewUrl(fileId);

      const whatsappUrl = generateWhatsAppLink(mobile, clientName, company, renewalDate, fileUrl, policyNo);
      const gmailUrl = generateGmailComposeLink(email, clientName, company, renewalDate, fileUrl, policyNo);

      sheet.getRange(row, CONFIG.COLS.WHATSAPP_LINK, 1, 2).setValues([[
        whatsappUrl ? '=HYPERLINK("' + whatsappUrl + '", "\uD83D\uDCAC Send WhatsApp")' : '\u2014',
        gmailUrl ? '=HYPERLINK("' + gmailUrl + '", "\uD83D\uDCE7 Send Gmail")' : '\u2014'
      ]]);
    }
  } catch (err) {
    console.error('onEdit trigger error: ' + err.toString());
  }
}

function setupSystem() {
  const ui = SpreadsheetApp.getUi();
  const props = PropertiesService.getScriptProperties();

  let apiKey = props.getProperty(CONFIG.PROP_GEMINI_KEY);
  if (!apiKey) {
    const response = ui.prompt(
      'Gemini API Key Required',
      'Please enter your Google Gemini API Key:\n(Get your free key at https://aistudio.google.com/app/apikey)',
      ui.ButtonSet.OK_CANCEL
    );
    if (response.getSelectedButton() !== ui.Button.OK) {
      ui.alert('Setup Cancelled', 'Setup was stopped because no Gemini API key was provided.', ui.ButtonSet.OK);
      return;
    }
    apiKey = response.getResponseText().trim();
    if (!apiKey) {
      ui.alert('Invalid API Key', 'The API key cannot be empty.', ui.ButtonSet.OK);
      return;
    }
    props.setProperty(CONFIG.PROP_GEMINI_KEY, apiKey);
  }

  props.deleteProperty(CONFIG.PROP_ACTIVE_MODEL);
  props.deleteProperty(CONFIG.PROP_MODELS_LIST);

  let masterFolder, inboxFolder, needsReviewFolder;

  const existingMasterId = props.getProperty(CONFIG.PROP_MASTER_FOLDER_ID);
  if (existingMasterId && folderExists(existingMasterId)) {
    masterFolder = DriveApp.getFolderById(existingMasterId);
  } else {
    masterFolder = DriveApp.createFolder(CONFIG.MASTER_FOLDER_NAME);
    props.setProperty(CONFIG.PROP_MASTER_FOLDER_ID, masterFolder.getId());
  }

  try {
    masterFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (e) {
    Logger.log('Master folder sharing notice: ' + e.toString());
  }

  const existingInboxId = props.getProperty(CONFIG.PROP_INBOX_FOLDER_ID);
  if (existingInboxId && folderExists(existingInboxId)) {
    inboxFolder = DriveApp.getFolderById(existingInboxId);
  } else {
    inboxFolder = getOrCreateSubFolder(masterFolder, CONFIG.INBOX_FOLDER_NAME);
    props.setProperty(CONFIG.PROP_INBOX_FOLDER_ID, inboxFolder.getId());
  }

  const existingReviewId = props.getProperty(CONFIG.PROP_NEEDS_REVIEW_FOLDER_ID);
  if (existingReviewId && folderExists(existingReviewId)) {
    needsReviewFolder = DriveApp.getFolderById(existingReviewId);
  } else {
    needsReviewFolder = getOrCreateSubFolder(masterFolder, CONFIG.NEEDS_REVIEW_FOLDER_NAME);
    props.setProperty(CONFIG.PROP_NEEDS_REVIEW_FOLDER_ID, needsReviewFolder.getId());
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  setupMasterDataSheet(ss);
  setupMonthlySummarySheet(ss);

  ui.alert('Setup Complete',
    'System Setup Completed Successfully!\n\n' +
    'Master Folder: "' + masterFolder.getName() + '"\n' +
    'Inbox Folder: "' + inboxFolder.getName() + '"\n' +
    'Needs Review Folder: "' + needsReviewFolder.getName() + '"\n\n' +
    'How to use:\n' +
    '1. Upload renewal notice PDFs into your INBOX folder.\n' +
    '2. Click "RENEWAL AUTOMATION" > "Process Inbox".\n' +
    '3. Open the Action Center sidebar or click WhatsApp/Gmail in each row.\n' +
    '4. All PDFs are set to "Anyone with link can view" automatically.',
    ui.ButtonSet.OK);
}

function folderExists(folderId) {
  try {
    return !DriveApp.getFolderById(folderId).isTrashed();
  } catch (e) {
    return false;
  }
}

/**
 * Finds or creates a subfolder. Cached per execution so a batch of 10 PDFs landing in
 * the same month folder costs ONE Drive lookup, not ten.
 */
function getOrCreateSubFolder(parentFolder, folderName) {
  const cacheKey = parentFolder.getId() + '::' + folderName;
  if (_FOLDER_CACHE[cacheKey]) return _FOLDER_CACHE[cacheKey];

  const folders = parentFolder.getFoldersByName(folderName);
  let folder;
  let isNew = false;
  if (folders.hasNext()) {
    folder = folders.next();
  } else {
    folder = parentFolder.createFolder(folderName);
    isNew = true;
  }

  if (isNew) {
    try { folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
  }

  _FOLDER_CACHE[cacheKey] = folder;
  return folder;
}

// 4. PARALLEL DRIVE ENGINE (REST + fetchAll, with DriveApp fallback)
// ======================================================================================

function getDriveToken() {
  try { return ScriptApp.getOAuthToken(); } catch (e) { return null; }
}

/**
 * Runs an array of UrlFetchApp request objects in parallel, chunked for safety.
 */
function safeFetchAll(requests, chunkSize) {
  if (!requests || requests.length === 0) return [];
  const size = chunkSize || 25;
  let out = [];
  for (let i = 0; i < requests.length; i += size) {
    const slice = requests.slice(i, i + size);
    try {
      out = out.concat(UrlFetchApp.fetchAll(slice));
    } catch (e) {
      Logger.log('fetchAll fallback: ' + e.toString());
      slice.forEach(function (r) {
        try { out.push(UrlFetchApp.fetch(r.url, r)); } catch (e2) { out.push(null); }
      });
    }
  }
  return out;
}

/**
 * Downloads all PDF bytes in PARALLEL (was 10 sequential file.getBlob() calls).
 */
function fetchFileBlobsParallel(items) {
  const token = getDriveToken();
  if (token && items.length > 1) {
    const reqs = items.map(function (it) {
      return {
        url: 'https://www.googleapis.com/drive/v3/files/' + it.fileId + '?alt=media&supportsAllDrives=true',
        method: 'get',
        headers: { Authorization: 'Bearer ' + token },
        muteHttpExceptions: true
      };
    });
    const res = safeFetchAll(reqs, 10);
    for (let i = 0; i < items.length; i++) {
      const r = res[i];
      if (r) {
        try { if (r.getResponseCode() === 200) items[i].blob = r.getBlob(); } catch (e) {}
      }
    }
  }
  items.forEach(function (it) {
    if (!it.blob) {
      try { it.blob = it.file.getBlob(); } catch (e) { it.blob = null; }
    }
  });
}

/**
 * Applies rename + move + public sharing for a whole batch in ONE parallel round.
 */
function applyDriveBatch(ops) {
  if (!ops || ops.length === 0) return;
  const token = getDriveToken();

  if (!token) {
    ops.forEach(applySingleDriveOpFallback);
    return;
  }

  const reqs = [];
  const map = [];

  ops.forEach(function (op) {
    const willMove = op.addParentId && op.addParentId !== op.removeParentId;
    if (op.newName || willMove) {
      const params = ['supportsAllDrives=true', 'fields=id'];
      if (willMove) {
        params.push('addParents=' + op.addParentId);
        if (op.removeParentId) params.push('removeParents=' + op.removeParentId);
      }
      const body = {};
      if (op.newName) body.name = op.newName;
      reqs.push({
        url: 'https://www.googleapis.com/drive/v3/files/' + op.fileId + '?' + params.join('&'),
        method: 'patch',
        contentType: 'application/json',
        headers: { Authorization: 'Bearer ' + token },
        payload: JSON.stringify(body),
        muteHttpExceptions: true
      });
      map.push({ op: op, kind: 'update' });
    }

    if (!_SHARE_CACHE[op.fileId]) {
      reqs.push({
        url: 'https://www.googleapis.com/drive/v3/files/' + op.fileId + '/permissions?supportsAllDrives=true&sendNotificationEmail=false&fields=id',
        method: 'post',
        contentType: 'application/json',
        headers: { Authorization: 'Bearer ' + token },
        payload: JSON.stringify({ role: 'reader', type: 'anyone' }),
        muteHttpExceptions: true
      });
      map.push({ op: op, kind: 'share' });
    }
  });

  if (reqs.length === 0) return;

  const res = safeFetchAll(reqs, 25);

  for (let i = 0; i < map.length; i++) {
    const entry = map[i];
    let code = 0;
    try { code = res[i] ? res[i].getResponseCode() : 0; } catch (e) {}

    if (code >= 200 && code < 300) {
      if (entry.kind === 'share') _SHARE_CACHE[entry.op.fileId] = true;
    } else {
      entry.op._needsFallback = true;
      Logger.log('Drive REST ' + entry.kind + ' returned ' + code + ' for ' + entry.op.fileId + ' - retrying via DriveApp.');
    }
  }

  ops.filter(function (o) { return o._needsFallback; }).forEach(applySingleDriveOpFallback);
}

function applySingleDriveOpFallback(op) {
  try {
    const file = op.file || DriveApp.getFileById(op.fileId);
    if (op.newName) { try { file.setName(op.newName); } catch (e) {} }
    if (op.targetFolder && op.addParentId !== op.removeParentId) {
      try { file.moveTo(op.targetFolder); } catch (e) {}
    }
    if (!_SHARE_CACHE[op.fileId]) {
      try {
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        _SHARE_CACHE[op.fileId] = true;
      } catch (e) {}
    }
  } catch (err) {
    Logger.log('Drive fallback failed for ' + op.fileId + ': ' + err.message);
  }
}

function bulkShareFileIds(fileIds) {
  const pending = fileIds.filter(function (id) { return id && !_SHARE_CACHE[id]; });
  if (pending.length === 0) return 0;

  const token = getDriveToken();
  let shared = 0;

  if (token) {
    const reqs = pending.map(function (id) {
      return {
        url: 'https://www.googleapis.com/drive/v3/files/' + id + '/permissions?supportsAllDrives=true&sendNotificationEmail=false&fields=id',
        method: 'post',
        contentType: 'application/json',
        headers: { Authorization: 'Bearer ' + token },
        payload: JSON.stringify({ role: 'reader', type: 'anyone' }),
        muteHttpExceptions: true
      };
    });
    const res = safeFetchAll(reqs, 25);
    for (let i = 0; i < pending.length; i++) {
      let code = 0;
      try { code = res[i] ? res[i].getResponseCode() : 0; } catch (e) {}
      if (code >= 200 && code < 300) { _SHARE_CACHE[pending[i]] = true; shared++; }
    }
  }

  pending.forEach(function (id) {
    if (!_SHARE_CACHE[id]) {
      try {
        DriveApp.getFileById(id).setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        _SHARE_CACHE[id] = true;
        shared++;
      } catch (e) {}
    }
  });

  return shared;
}

// 5. PDF TEXT EXTRACTION (WITH GARBAGE GUARD)
// ======================================================================================

/**
 * Fast PDF text stream parser.
 *
 * CRITICAL FIX: many insurer PDFs use compressed / subset-embedded fonts. Scraping their
 * raw streams produces thousands of characters of noise like "* Z< xSW 3 } } [ m ][C 1 5 J".
 * The old code treated that as valid text and sent it to Gemini, which then returned
 * nothing ("Extraction payload was empty"). We now score the result and reject it unless
 * it genuinely looks like prose, in which case the full PDF is sent instead.
 */
function extractTextFromPdfBlob(blob) {
  try {
    if (!blob) return null;
    const bytes = blob.getBytes();
    if (!bytes || bytes.length === 0) return null;

    const scanLen = Math.min(bytes.length, 900000);
    const slice = (bytes.length > scanLen) ? bytes.slice(0, scanLen) : bytes;
    const rawStr = Utilities.newBlob(slice).getDataAsString('ISO-8859-1');

    const chunks = [];

    // Pattern 1: literal text in parentheses, e.g. (Policyholder Name)
    const matches1 = rawStr.match(/\(([^()]{2,160})\)/g);
    if (matches1) {
      for (let i = 0; i < matches1.length; i++) {
        const clean = matches1[i].slice(1, -1)
          .replace(/\\([0-7]{1,3})/g, function (m, oct) { return String.fromCharCode(parseInt(oct, 8)); })
          .replace(/\\[rnbtf]/g, ' ')
          .replace(/\\([()])/g, '$1')
          .replace(/\\\\/g, '\\')
          .replace(/[^\x20-\x7E]/g, ' ')
          .replace(/\s+/g, ' ')
          .trim();
        if (clean.length > 1 && /[a-zA-Z0-9]/.test(clean)) chunks.push(clean);
      }
    }

    // Pattern 2: hex-encoded text chunks, e.g. <416A6179>
    if (chunks.length < 15) {
      const matches2 = rawStr.match(/<([0-9A-Fa-f]{8,120})>/g);
      if (matches2) {
        for (let i = 0; i < matches2.length; i++) {
          const hex = matches2[i].slice(1, -1);
          let str = '';
          for (let k = 0; k < hex.length - 1; k += 2) {
            const code = parseInt(hex.substr(k, 2), 16);
            if (code >= 32 && code <= 126) str += String.fromCharCode(code);
          }
          if (str.length > 2 && /[a-zA-Z0-9]/.test(str)) chunks.push(str);
        }
      }
    }

    if (chunks.length >= 8) {
      const textClean = chunks.join(' ').replace(/\s+/g, ' ').trim();
      const quality = scoreExtractedText(textClean);
      if (quality.ok) return textClean.substring(0, 4500);
      Logger.log('Text extraction rejected as stream noise (' + quality.why + ') - sending full PDF instead.');
    }
  } catch (e) {
    Logger.log('Text extraction notice: ' + e.toString());
  }
  return null;
}

/**
 * Decides whether scraped text is real document text or compressed-stream garbage.
 */
function scoreExtractedText(text) {
  if (!text || text.length < 80) return { ok: false, why: 'too short' };
  const lower = text.toLowerCase();
  const keys = ['policy', 'premium', 'renewal', 'insurance', 'expiry', 'customer', 'address'];
  let hits = 0;
  for (let i = 0; i < keys.length; i++) {
    if (lower.indexOf(keys[i]) !== -1) hits++;
  }
  if (hits < 2) return { ok: false, why: 'compressed stream, not real text (' + hits + ' keywords)' };
  return { ok: true, why: 'ok (' + hits + ' keywords)' };
}

function buildGeminiRequest(blob, apiKey, model, forcePdf, cachedB64) {
  const fileName = (blob && blob.getName) ? String(blob.getName() || '') : '';
  const digitalText = extractTextFromPdfBlob(blob) || '';
  const rules =
    'Return ONLY JSON with keys clientName, renewalDate, insuranceCompany, policyNumber, clientMobile, clientEmail.\n' +
    'clientName is the person after "Name :". Do not use the intermediary (agent) name and do not use family member names.\n' +
    'renewalDate is the Policy Expiry Date as YYYY-MM-DD, not the letter date.\n' +
    'insuranceCompany is the insurer. policyNumber is Policy No. Mobile and email may be blank.\n' +
    'File name: ' + fileName;

  if (digitalText) {
    return {
      mode: 'text',
      b64: '',
      textLen: digitalText.length,
      req: {
        url: 'https://generativelanguage.googleapis.com/' + CONFIG.GEMINI_API_VERSION +
             '/models/' + model + ':generateContent',
        method: 'post',
        contentType: 'application/json',
        headers: { 'x-goog-api-key': apiKey },
        payload: JSON.stringify({
          contents: [{ parts: [{ text: rules + '\n\n' + digitalText }] }]
        }),
        muteHttpExceptions: true
      }
    };
  }

  let b64 = cachedB64 || '';
  if (!b64 && blob) b64 = Utilities.base64Encode(blob.getBytes());
  return {
    mode: 'pdf',
    b64: b64,
    textLen: 0,
    req: {
      url: 'https://generativelanguage.googleapis.com/' + CONFIG.GEMINI_API_VERSION +
           '/models/' + model + ':generateContent',
      method: 'post',
      contentType: 'application/json',
      headers: { 'x-goog-api-key': apiKey },
      payload: JSON.stringify({
        contents: [{ parts: [
          { text: rules },
          { inlineData: { mimeType: 'application/pdf', data: b64 } }
        ] }]
      }),
      muteHttpExceptions: true
    }
  };
}

function prepareGeminiRequest(fileOrBlob, apiKey, model) {
  const blob = (fileOrBlob && typeof fileOrBlob.getBlob === 'function') ? fileOrBlob.getBlob() : fileOrBlob;
  return buildGeminiRequest(blob, apiKey, model, false).req;
}

/**
 * Parses a Gemini HTTP response. Returns { data, httpCode, reason }.
 */
function parseGeminiResponse(resp) {
  const result = { data: null, httpCode: 0, reason: '' };
  if (!resp) { result.reason = 'no response'; return result; }

  try {
    result.httpCode = resp.getResponseCode();
  } catch (e) {
    result.reason = 'unreadable response';
    return result;
  }

  if (result.httpCode === 503) { result.reason = 'model overloaded (HTTP 503)'; return result; }
  if (result.httpCode === 429) { result.reason = 'rate limit hit (HTTP 429)'; return result; }
  if (result.httpCode === 404) {
    let extra = '';
    try { extra = resp.getContentText(); } catch (e) {}
    if (extra.indexOf('no longer available') !== -1) {
      result.reason = 'model not available for this API key (HTTP 404)';
    } else {
      result.reason = 'model not found (HTTP 404)';
    }
    return result;
  }
  if (result.httpCode !== 200) {
    var body = '';
    try { body = String(resp.getContentText() || ''); } catch (e2) {}
    var msg = '';
    try { msg = String((JSON.parse(body).error || {}).message || ''); } catch (e3) {}
    result.reason = ('HTTP ' + result.httpCode + (msg ? (': ' + msg) : '')).substring(0, 220);
    return result;
  }

  try {
    const json = JSON.parse(resp.getContentText());
    if (!json.candidates || json.candidates.length === 0) {
      result.reason = 'no candidates returned';
      return result;
    }
    const cand = json.candidates[0];
    if (cand.finishReason === 'MAX_TOKENS') { result.reason = 'response truncated (MAX_TOKENS)'; return result; }
    if (cand.finishReason === 'SAFETY') { result.reason = 'blocked by safety filter'; return result; }
    if (!cand.content || !cand.content.parts || cand.content.parts.length === 0) {
      result.reason = 'empty content block';
      return result;
    }

    let str = '';
    const partsOut = cand.content.parts;
    for (let p = 0; p < partsOut.length; p++) {
      if (partsOut[p].thought) continue;
      if (partsOut[p].text) { str = String(partsOut[p].text).trim(); break; }
    }
    if (str.indexOf('```json') === 0) str = str.replace(/^```json\s*/, '').replace(/\s*```$/, '');
    else if (str.indexOf('```') === 0) str = str.replace(/^```\s*/, '').replace(/\s*```$/, '');

    const parsed = JSON.parse(str);
    if (parsed && parsed.renewalDate) parsed.renewalDate = normalizeDateToYYYYMMDD(parsed.renewalDate);
    result.data = parsed;
  } catch (e) {
    result.reason = 'could not parse reply: ' + e.message;
  }
  return result;
}

/** True when a second same-model call is worth the extra seconds. */
function isTransientGeminiFailure(reason) {
  if (!reason) return false;
  const r = String(reason);
  return r.indexOf('503') !== -1 ||
         r.indexOf('429') !== -1 ||
         r.indexOf('empty content') !== -1 ||
         r.indexOf('no candidates') !== -1 ||
         r.indexOf('no response') !== -1 ||
         r.indexOf('unreadable') !== -1 ||
         r.indexOf('HTTP 5') !== -1 ||
         r.indexOf('could not parse') !== -1;
}

function processBatchConcurrent(filesToProcess, masterFolder, reviewFolder, masterSheet, existingRecords, apiKey, isNeedsReviewMode, sourceFolderId) {
  const startTime = Date.now();

  const stats = {
    totalFound: filesToProcess ? filesToProcess.length : 0,
    processed: 0, success: 0, needsReview: 0, skipped: 0, remaining: 0,
    overloaded: 0, elapsedSeconds: '0.0', fileDetails: []
  };

  if (!filesToProcess || filesToProcess.length === 0) return stats;

  const model = resolveWorkingGeminiModel(apiKey);

  // ---- Step 1: filter duplicates ---------------------------------------------------
  const items = [];
  for (let i = 0; i < filesToProcess.length; i++) {
    const file = filesToProcess[i];
    const fileId = file.getId();

    if (!isNeedsReviewMode && existingRecords.has(fileId)) {
      if (existingRecords.get(fileId).status === 'SUCCESS') { stats.skipped++; continue; }
    }

    items.push({
      file: file, fileId: fileId, originalName: file.getName(),
      globalIndex: i + 1, blob: null, b64: '', mode: 'pdf', extraction: null, reason: ''
    });
  }

  if (items.length === 0) {
    stats.elapsedSeconds = ((Date.now() - startTime) / 1000).toFixed(1);
    return stats;
  }
  stats.processed = items.length;

  Logger.log('Batch start: ' + items.length + ' files | model: ' + model);

  // ---- Step 2: parallel download ---------------------------------------------------
  let t = Date.now();
  fetchFileBlobsParallel(items);
  Logger.log('Parallel download: ' + ((Date.now() - t) / 1000).toFixed(2) + 's');

  // ---- Step 3: one file at a time on the model that is actually up ----
  t = Date.now();
  const active = [];
  const candidates = listGeminiCandidates();
  let usedModel = model;

  items.forEach(function (it) {
    if (!it.blob) { it.reason = 'PDF content could not be read from Drive'; return; }
    active.push(it);
    let done = false;
    for (let m = 0; m < candidates.length && !done; m++) {
      const tryModel = candidates[m];
      for (let attempt = 0; attempt < 3 && !done; attempt++) {
        if (Date.now() - startTime > CONFIG.MAX_RUNTIME_MS) break;
        if (attempt > 0) Utilities.sleep(CONFIG.RETRY_PAUSE_MS * attempt);
        const built = buildGeminiRequest(it.blob, apiKey, tryModel, false);
        let resp = null;
        try {
          resp = UrlFetchApp.fetch(built.req.url, built.req);
        } catch (e) {
          it.reason = e.message + ' [' + tryModel + ']';
          continue;
        }
        const r = parseGeminiResponse(resp);
        if (r.data) {
          it.extraction = r.data;
          it.reason = '';
          usedModel = tryModel;
          _ACTIVE_WORKING_MODEL = tryModel;
          done = true;
          break;
        }
        it.reason = (r.reason || 'empty') + ' [' + tryModel + ']';
        const badModel = it.reason.indexOf('404') !== -1 || it.reason.indexOf('400') !== -1;
        if (badModel) break;
        if (!isTransientGeminiFailure(it.reason)) break;
      }
    }
    Utilities.sleep(1500);
  });
  Logger.log('Gemini round: ' + ((Date.now() - t) / 1000).toFixed(2) + 's using ' + usedModel);

  items.forEach(function (it) { it.blob = null; it.b64 = ''; });

  // ---- Step 5: parallel rename + move + share --------------------------------------
  const driveOps = [];
  items.forEach(function (it) {
    const validation = validateExtraction(it.extraction);
    it.valid = validation.isValid;
    if (!it.valid && !it.reason) it.reason = validation.reason;
    if (!it.valid && it.reason && it.reason.indexOf('503') !== -1) stats.overloaded++;

    if (it.valid) {
      const renewalDate = it.extraction.renewalDate;
      it.monthYear = formatMonthYear(renewalDate);
      it.targetFolder = getOrCreateSubFolder(masterFolder, it.monthYear);
      it.newName = renewalDate + '-' + cleanClientNameForFilename(it.extraction.clientName) +
                   '-' + cleanCompanyName(it.extraction.insuranceCompany) + '.pdf';
    } else {
      it.monthYear = (it.extraction && it.extraction.renewalDate) ? formatMonthYear(it.extraction.renewalDate) : 'UNKNOWN';
      it.targetFolder = reviewFolder;
      it.newName = null;
    }

    driveOps.push({
      fileId: it.fileId, file: it.file, newName: it.newName, targetFolder: it.targetFolder,
      addParentId: it.targetFolder.getId(), removeParentId: sourceFolderId || null
    });
  });

  t = Date.now();
  applyDriveBatch(driveOps);
  Logger.log('Parallel Drive ops: ' + ((Date.now() - t) / 1000).toFixed(2) + 's');

  // ---- Step 6: bulk sheet write ----------------------------------------------------
  const timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  let nextSrNo = getNextSerialNumber(masterSheet);

  const appendRows = [];
  const updateRows = [];

  items.forEach(function (it) {
    const existingRowMeta = existingRecords.get(it.fileId);
    const fileUrl = buildDriveViewUrl(it.fileId);
    const folderUrl = it.targetFolder.getUrl();
    const ext = it.extraction || {};

    let rowData;
    if (it.valid) {
      rowData = {
        srNo: existingRowMeta ? existingRowMeta.srNo : nextSrNo++,
        clientName: ext.clientName, renewalDate: ext.renewalDate, month: it.monthYear,
        company: ext.insuranceCompany, policyNumber: ext.policyNumber || '',
        clientMobile: ext.clientMobile || '', clientEmail: ext.clientEmail || '',
        whatsappUrl: generateWhatsAppLink(ext.clientMobile, ext.clientName, ext.insuranceCompany, ext.renewalDate, fileUrl, ext.policyNumber),
        gmailUrl: generateGmailComposeLink(ext.clientEmail, ext.clientName, ext.insuranceCompany, ext.renewalDate, fileUrl, ext.policyNumber),
        fileName: it.newName, fileUrl: fileUrl, folderName: it.monthYear, folderUrl: folderUrl,
        status: 'SUCCESS', dateProcessed: timestamp,
        originalName: (isNeedsReviewMode && existingRowMeta) ? existingRowMeta.originalName : it.originalName,
        fileId: it.fileId, notes: isNeedsReviewMode ? 'Resolved via Needs Review reprocessing' : ''
      };
      stats.success++;
      stats.fileDetails.push({ name: it.originalName, status: 'SUCCESS', client: ext.clientName, date: ext.renewalDate });
    } else {
      rowData = {
        srNo: existingRowMeta ? existingRowMeta.srNo : nextSrNo++,
        clientName: ext.clientName || 'UNKNOWN', renewalDate: ext.renewalDate || '', month: it.monthYear,
        company: ext.insuranceCompany || 'UNKNOWN', policyNumber: ext.policyNumber || '',
        clientMobile: ext.clientMobile || '', clientEmail: ext.clientEmail || '',
        whatsappUrl: '', gmailUrl: '',
        fileName: it.originalName, fileUrl: fileUrl,
        folderName: CONFIG.NEEDS_REVIEW_FOLDER_NAME, folderUrl: folderUrl,
        status: 'NEEDS REVIEW', dateProcessed: timestamp,
        originalName: (isNeedsReviewMode && existingRowMeta) ? existingRowMeta.originalName : it.originalName,
        fileId: it.fileId, notes: it.reason || 'Validation failed'
      };
      stats.needsReview++;
      stats.fileDetails.push({ name: it.originalName, status: 'NEEDS REVIEW', reason: it.reason });
    }

    const values = buildMasterRowValues(rowData);
    if (existingRowMeta && existingRowMeta.rowIndex) {
      updateRows.push({ rowIndex: existingRowMeta.rowIndex, values: values, status: rowData.status });
    } else {
      appendRows.push({ values: values, status: rowData.status });
    }
  });

  t = Date.now();
  flushMasterRows(masterSheet, appendRows, updateRows);
  Logger.log('Bulk sheet write: ' + ((Date.now() - t) / 1000).toFixed(2) + 's');

  refreshMonthlySummary();

  stats.elapsedSeconds = ((Date.now() - startTime) / 1000).toFixed(1);
  Logger.log('Batch complete in ' + stats.elapsedSeconds + 's - Success: ' + stats.success +
             ', Needs Review: ' + stats.needsReview + ', Skipped: ' + stats.skipped);
  return stats;
}

  const props = PropertiesService.getScriptProperties();
  const inboxId = props.getProperty(CONFIG.PROP_INBOX_FOLDER_ID);
  const masterId = props.getProperty(CONFIG.PROP_MASTER_FOLDER_ID);
  const reviewId = props.getProperty(CONFIG.PROP_NEEDS_REVIEW_FOLDER_ID);
  const apiKey = props.getProperty(CONFIG.PROP_GEMINI_KEY);

  if (!inboxId || !masterId || !reviewId) {
    throw new Error('System is not initialized. Please run "Setup / Initialize System" first.');
  }
  if (!apiKey) {
    throw new Error('Gemini API Key is missing. Please set it using "Set / Update Gemini API Key".');
  }

  const inboxFolder = DriveApp.getFolderById(inboxId);
  const masterFolder = DriveApp.getFolderById(masterId);
  const reviewFolder = DriveApp.getFolderById(reviewId);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const masterSheet = ss.getSheetByName(CONFIG.SHEET_MASTER_DATA) || setupMasterDataSheet(ss);
  ensureMasterDataSchema(masterSheet);
  const existingRecords = getExistingFileMap(masterSheet);

  const filesIter = inboxFolder.getFiles();
  const pdfFiles = [];
  while (filesIter.hasNext()) {
    const file = filesIter.next();
    if (file.getMimeType() === MimeType.PDF || file.getName().toLowerCase().indexOf('.pdf') !== -1) {

    }
  }

  const filesToProcess = pdfFiles.slice(0, CONFIG.MAX_FILES_PER_RUN);
  const stats = processBatchConcurrent(filesToProcess, masterFolder, reviewFolder, masterSheet, existingRecords, apiKey, false, inboxId);
  stats.totalFound = pdfFiles.length;
  stats.remaining = Math.max(0, pdfFiles.length - filesToProcess.length);
  return stats;
}

function processNeedsReview() {
  const props = PropertiesService.getScriptProperties();
  const masterId = props.getProperty(CONFIG.PROP_MASTER_FOLDER_ID);
  const reviewId = props.getProperty(CONFIG.PROP_NEEDS_REVIEW_FOLDER_ID);
  const apiKey = props.getProperty(CONFIG.PROP_GEMINI_KEY);

  if (!masterId || !reviewId) {
    throw new Error('System is not initialized. Please run "Setup / Initialize System" first.');
  }
  if (!apiKey) {
    throw new Error('Gemini API Key is missing. Please set it using "Set / Update Gemini API Key".');
  }

  const masterFolder = DriveApp.getFolderById(masterId);
  const reviewFolder = DriveApp.getFolderById(reviewId);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const masterSheet = ss.getSheetByName(CONFIG.SHEET_MASTER_DATA) || setupMasterDataSheet(ss);
  ensureMasterDataSchema(masterSheet);
  const existingRecords = getExistingFileMap(masterSheet);

  const filesIter = reviewFolder.getFiles();
  const pdfFiles = [];
  while (filesIter.hasNext()) {
    const file = filesIter.next();
    if (file.getMimeType() === MimeType.PDF || file.getName().toLowerCase().indexOf('.pdf') !== -1) {
      pdfFiles.push(file);
    }
  }

  const filesToProcess = pdfFiles.slice(0, CONFIG.MAX_FILES_PER_RUN);
  const stats = processBatchConcurrent(filesToProcess, masterFolder, reviewFolder, masterSheet, existingRecords, apiKey, true, reviewId);
  stats.totalFound = pdfFiles.length;
  stats.remaining = Math.max(0, pdfFiles.length - filesToProcess.length);
  stats.stillNeedsReview = stats.needsReview;
  return stats;
}

function extractDriveFileId(input) {
  if (!input) return '';
  const str = String(input).trim();
  if (/^[a-zA-Z0-9_-]{25,}$/.test(str) && str.indexOf('http') === -1 && str.indexOf('/') === -1 && str.indexOf('"') === -1) {
    return str;
  }
  const match = str.match(/\/d\/([a-zA-Z0-9_-]{25,})/i) || str.match(/[?&]id=([a-zA-Z0-9_-]{25,})/i);
  return match ? match[1] : '';
}

/** Pure string builder - zero API calls. */
function buildDriveViewUrl(fileId) {
  return fileId ? ('https://drive.google.com/file/d/' + fileId + '/view?usp=sharing') : '';
}

/**
 * Returns the public sharing URL for a file.
 * PERFORMANCE: no longer makes a Drive API call on every invocation. Permissions are
 * granted once per file during processing, or via the bulk-share menu item.
 */
function getPublicShareUrl(fileOrId) {
  if (!fileOrId) return '';
  try {
    if (typeof fileOrId === 'string') {
      const id = extractDriveFileId(fileOrId);
      if (id) return buildDriveViewUrl(id);
      return (fileOrId.indexOf('http') === 0) ? fileOrId : '';
    }
    return buildDriveViewUrl(fileOrId.getId());
  } catch (err) {
    return '';
  }
}

function generateWhatsAppLink(phone, clientName, company, renewalDate, fileUrl, policyNumber) {
  let cleanPhone = String(phone || '').replace(/[^0-9]/g, '');
  if (cleanPhone.indexOf('0') === 0 && cleanPhone.length === 11) {
    cleanPhone = '91' + cleanPhone.substring(1);
  } else if (cleanPhone.length === 10) {
    cleanPhone = '91' + cleanPhone;
  }

  const policyText = policyNumber ? (' (Policy No: ' + policyNumber + ')') : '';
  const cleanFileUrl = getPublicShareUrl(fileUrl);

  const msg =
    'Dear ' + (clientName || 'Valued Customer') + ',\n\n' +
    'This is a gentle reminder that your ' + (company || 'Insurance') + ' policy' + policyText +
    ' is due for renewal on *' + renewalDate + '*.\n\n' +
    'Kindly renew your policy on time to enjoy uninterrupted coverage, continuous claim benefits, and bonus protections.\n\n' +
    (cleanFileUrl ? ('\uD83D\uDCC4 View Renewal Notice: ' + cleanFileUrl + '\n\n') : '') +
    'Please reach out to me if you need any assistance with the payment link or policy updates.\n\n' +
    'Best Regards,\n' +
    '*' + CONFIG.SENDER_NAME + '*\n' +
    '\uD83C\uDF10 ' + CONFIG.SENDER_WEBSITE + '\n' +
    '\uD83D\uDCDE ' + CONFIG.SENDER_PHONE;

  const encoded = encodeURIComponent(msg);
  if (cleanPhone && cleanPhone.length >= 10) {
    return 'https://api.whatsapp.com/send?phone=' + cleanPhone + '&text=' + encoded;
  }
  return 'https://api.whatsapp.com/send?text=' + encoded;
}

function generateGmailComposeLink(email, clientName, company, renewalDate, fileUrl, policyNumber) {
  const cleanEmail = String(email || '').trim();
  const policyText = policyNumber ? (' (Policy No: ' + policyNumber + ')') : '';

    'Best Regards,\n' + CONFIG.SENDER_NAME + '\n' + CONFIG.SENDER_WEBSITE + '\n' +

/**
 * One-click bulk share. Permission grants run in parallel; the sheet is written in
 * bulk column ranges instead of 3 setFormula calls per row.
 */
function makeAllFilesPubliclyViewable() {
  const ui = SpreadsheetApp.getUi();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.SHEET_MASTER_DATA);
  const props = PropertiesService.getScriptProperties();
  const masterId = props.getProperty(CONFIG.PROP_MASTER_FOLDER_ID);

  const allIds = new Set();
  let sheetCount = 0;

  if (sheet) {
    const lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      const numRows = lastRow - 1;
      const data = sheet.getRange(2, 1, numRows, CONFIG.COLS.NOTES).getValues();
      const formulas = sheet.getRange(2, 1, numRows, CONFIG.COLS.NOTES).getFormulas();

      const actionCols = [];
      const fileLinkCol = [];

      for (let i = 0; i < numRows; i++) {
        const row = data[i];
        const fileFormula = String(formulas[i][CONFIG.COLS.FILE_LINK - 1] || '');
        const fileId = extractDriveFileId(String(row[CONFIG.COLS.FILE_ID - 1] || '').trim()) || extractDriveFileId(fileFormula);

        if (!fileId) {
          actionCols.push([
            formulas[i][CONFIG.COLS.WHATSAPP_LINK - 1] || String(row[CONFIG.COLS.WHATSAPP_LINK - 1] || '\u2014'),
            formulas[i][CONFIG.COLS.EMAIL_ACTION - 1] || String(row[CONFIG.COLS.EMAIL_ACTION - 1] || '\u2014')
          ]);
          fileLinkCol.push([fileFormula || String(row[CONFIG.COLS.FILE_LINK - 1] || '')]);
          continue;
        }

        allIds.add(fileId);
        sheetCount++;

        const publicUrl = buildDriveViewUrl(fileId);
        const waUrl = generateWhatsAppLink(
          String(row[CONFIG.COLS.CLIENT_MOBILE - 1] || ''), String(row[CONFIG.COLS.CLIENT_NAME - 1] || ''),
          String(row[CONFIG.COLS.COMPANY - 1] || ''), String(row[CONFIG.COLS.RENEWAL_DATE - 1] || ''),
          publicUrl, String(row[CONFIG.COLS.POLICY_NO - 1] || ''));
        const gmUrl = generateGmailComposeLink(
          String(row[CONFIG.COLS.CLIENT_EMAIL - 1] || ''), String(row[CONFIG.COLS.CLIENT_NAME - 1] || ''),
          String(row[CONFIG.COLS.COMPANY - 1] || ''), String(row[CONFIG.COLS.RENEWAL_DATE - 1] || ''),
          publicUrl, String(row[CONFIG.COLS.POLICY_NO - 1] || ''));

        actionCols.push([
          '=HYPERLINK("' + waUrl + '", "\uD83D\uDCAC Send WhatsApp")',
          '=HYPERLINK("' + gmUrl + '", "\uD83D\uDCE7 Send Gmail")'
        ]);
        fileLinkCol.push(['=HYPERLINK("' + publicUrl + '", "\uD83D\uDCC4 Open PDF")']);
      }

      sheet.getRange(2, CONFIG.COLS.WHATSAPP_LINK, numRows, 2).setValues(actionCols);
      sheet.getRange(2, CONFIG.COLS.FILE_LINK, numRows, 1).setValues(fileLinkCol);
    }
  }

  if (masterId && folderExists(masterId)) {
    try { collectFolderFileIds(DriveApp.getFolderById(masterId), allIds); }
    catch (e) { Logger.log('Drive folder recursion error: ' + e.message); }
  }

  const idArray = Array.from(allIds);
  bulkShareFileIds(idArray);

  ui.alert('Sharing Permissions Updated',
    'Direct Viewing Rights Enabled!\n\n' +
    '\u2022 Total Renewal Notices Unlocked: ' + idArray.length + '\n' +
    '\u2022 Master Sheet Links Updated: ' + sheetCount + '\n\n' +
    'Clients opening notice links will now view the PDF directly without a "Request Access" screen.',
    ui.ButtonSet.OK);
}

  const clientName = String(rowData[CONFIG.COLS.CLIENT_NAME - 1] || '');

    const renDate = new Date(renewalDate);
    if (!isNaN(renDate.getTime())) {
      const today = new Date();
      today.setHours(0, 0, 0, 0);
      renDate.setHours(0, 0, 0, 0);
      daysRemaining = Math.round((renDate - today) / (1000 * 60 * 60 * 24));
    }
  }

  return {
    rowIndex: row, clientName: clientName, renewalDate: renewalDate, company: company,
    policyNo: policyNo, mobile: mobile, email: email,
    status: String(rowData[CONFIG.COLS.STATUS - 1] || ''),
    fileUrl: fileUrl, fileId: fileId,
    whatsappUrl: generateWhatsAppLink(mobile, clientName, company, renewalDate, fileUrl, policyNo),
    gmailUrl: generateGmailComposeLink(email, clientName, company, renewalDate, fileUrl, policyNo),
    daysRemaining: daysRemaining
  };
}

function normalizeDateToYYYYMMDD(dateStr) {
  if (!dateStr) return '';
  const trimmed = String(dateStr).trim();

  if (/^\d{4}-\d{2}-\d{2}$/.test(trimmed)) return trimmed;

  const dmy = trimmed.match(/^(\d{1,2})[\/\-\.](\d{1,2})[\/\-\.](\d{4})$/);
  if (dmy) return dmy[3] + '-' + dmy[2].padStart(2, '0') + '-' + dmy[1].padStart(2, '0');

  const ymd = trimmed.match(/^(\d{4})[\/\-\.](\d{1,2})[\/\-\.](\d{1,2})$/);
  if (ymd) return ymd[1] + '-' + ymd[2].padStart(2, '0') + '-' + ymd[3].padStart(2, '0');

  const ts = Date.parse(trimmed.replace(/-/g, ' '));
  if (!isNaN(ts)) {
    const d = new Date(ts);
    return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
  }
  return trimmed;
}

function validateExtraction(extraction) {
  if (!extraction) return { isValid: false, reason: 'Extraction payload was empty' };

  const name = String(extraction.clientName || '').trim();
  const date = String(extraction.renewalDate || '').trim();
  const company = String(extraction.insuranceCompany || '').trim();

  if (!name || name.toLowerCase() === 'unknown' || name.length < 2) {
    return { isValid: false, reason: 'Client Name could not be identified' };
  }
  if (!date || !/^\d{4}-\d{2}-\d{2}$/.test(date)) {
    return { isValid: false, reason: 'Renewal Date missing or invalid format: "' + date + '"' };
  }
  if (isNaN(new Date(date).getTime())) {
    return { isValid: false, reason: 'Renewal Date "' + date + '" is not a valid calendar date' };
  }
  if (!company || company.toLowerCase() === 'unknown') {
    return { isValid: false, reason: 'Insurance Company could not be identified' };
  }
  return { isValid: true, reason: '' };
}

function cleanCompanyName(rawCompany) {
  if (!rawCompany) return 'INSURANCE';
  const lower = rawCompany.toLowerCase().trim();
  for (const key in COMPANY_MAP) {
    if (lower.indexOf(key) !== -1) return COMPANY_MAP[key];
  }
  return rawCompany.replace(/[^a-zA-Z0-9]/g, '_').replace(/_+/g, '_')
    .replace(/^_|_$/g, '').toUpperCase() || 'INSURANCE';
}

function cleanClientNameForFilename(rawName) {
  if (!rawName) return 'Client';
  return rawName
    .replace(/^(mr|mrs|ms|dr|prof|shri|smt)\.?\s+/i, '')
    .replace(/[^a-zA-Z0-9\s]/g, '')
    .trim().split(/\s+/)
    .map(function (w) { return w.charAt(0).toUpperCase() + w.slice(1).toLowerCase(); })
    .join('_') || 'Client';
}

function formatMonthYear(dateStr) {
  if (!dateStr || !/^\d{4}-\d{2}-\d{2}$/.test(dateStr)) return 'UNKNOWN';
  const parts = dateStr.split('-');
  const monthNames = ['January', 'February', 'March', 'April', 'May', 'June',
    'July', 'August', 'September', 'October', 'November', 'December'];
  return monthNames[parseInt(parts[1], 10) - 1] + ' ' + parts[0];
}

/** Applies all row formatting in bulk ranges (was ~15 calls PER ROW). */
function formatMasterRange(sheet, startRow, numRows, statuses) {
  if (numRows <= 0) return;

  sheet.getRange(startRow, 1, numRows, CONFIG.COLS.NOTES)
    .setFontFamily('Roboto').setFontSize(9).setVerticalAlignment('middle');

  sheet.setRowHeights(startRow, numRows, 26);

  [CONFIG.COLS.SR_NO, CONFIG.COLS.RENEWAL_DATE, CONFIG.COLS.FILE_LINK,
   CONFIG.COLS.FOLDER_LINK, CONFIG.COLS.STATUS, CONFIG.COLS.DATE_PROCESSED
  ].forEach(function (col) {
    sheet.getRange(startRow, col, numRows, 1).setHorizontalAlignment('center');
  });

  sheet.getRange(startRow, CONFIG.COLS.WHATSAPP_LINK, numRows, 1)
    .setHorizontalAlignment('center').setFontWeight('bold').setFontColor('#0F5132');
  sheet.getRange(startRow, CONFIG.COLS.EMAIL_ACTION, numRows, 1)
    .setHorizontalAlignment('center').setFontWeight('bold').setFontColor('#1E40AF');

  const bg = [];
  const fc = [];
  statuses.forEach(function (s) {
    if (s === 'SUCCESS') { bg.push(['#D1E7DD']); fc.push(['#0F5132']); }
    else if (s === 'NEEDS REVIEW') { bg.push(['#FFF3CD']); fc.push(['#664D03']); }
    else { bg.push(['#F8D7DA']); fc.push(['#842029']); }
  });
  sheet.getRange(startRow, CONFIG.COLS.STATUS, numRows, 1)
    .setBackgrounds(bg).setFontColors(fc).setFontWeight('bold');
}

/** Writes all new rows in ONE setValues call, then formats the block at once. */
function flushMasterRows(sheet, appendRows, updateRows) {
  if (appendRows && appendRows.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, appendRows.length, MASTER_HEADERS.length)
      .setValues(appendRows.map(function (r) { return r.values; }));
    formatMasterRange(sheet, startRow, appendRows.length,
      appendRows.map(function (r) { return r.status; }));
  }

  if (updateRows && updateRows.length > 0) {
    updateRows.forEach(function (u) {
      sheet.getRange(u.rowIndex, 1, 1, MASTER_HEADERS.length).setValues([u.values]);
      formatMasterRange(sheet, u.rowIndex, 1, [u.status]);
    });
  }
}

function appendOrUpdateMasterRow(sheet, existingMeta, rowData) {
  const values = buildMasterRowValues(rowData);
  if (existingMeta && existingMeta.rowIndex) {
    flushMasterRows(sheet, [], [{ rowIndex: existingMeta.rowIndex, values: values, status: rowData.status }]);
  } else {
    flushMasterRows(sheet, [{ values: values, status: rowData.status }], []);
  }
}

function migrateMasterDataSchema(sheet) {
  const lastRow = sheet.getLastRow();
  const oldCols = sheet.getLastColumn();
  if (lastRow <= 0 || oldCols <= 0) return;

  const newRows = [];
  const statuses = [];
  const idsToShare = [];

  if (lastRow >= 2) {
    const oldData = sheet.getRange(2, 1, lastRow - 1, oldCols).getValues();
    const oldFormulas = sheet.getRange(2, 1, lastRow - 1, oldCols).getFormulas();

    for (let i = 0; i < oldData.length; i++) {
      const row = oldData[i];
      const formRow = oldFormulas[i];

      const srNo = row[0] || (i + 1);
      const clientName = String(row[1] || '');
      const renewalDate = String(row[2] || '');
      const month = String(row[3] || '');
      const company = String(row[4] || '');

      let policyNumber = '', clientMobile = '', clientEmail = '', fileName = '';
      let fileLinkFormula = '', folderName = '', folderLinkFormula = '';
      let status = 'SUCCESS', dateProcessed = '', originalName = '', fileId = '', notes = '';

      if (oldCols >= 18) {
        policyNumber = String(row[CONFIG.COLS.POLICY_NO - 1] || '');
        clientMobile = String(row[CONFIG.COLS.CLIENT_MOBILE - 1] || '');
        clientEmail = String(row[CONFIG.COLS.CLIENT_EMAIL - 1] || '');
        fileName = String(row[CONFIG.COLS.FILE_NAME - 1] || '');
        fileLinkFormula = formRow[CONFIG.COLS.FILE_LINK - 1] || '';
        folderName = String(row[CONFIG.COLS.FOLDER_NAME - 1] || '');
        folderLinkFormula = formRow[CONFIG.COLS.FOLDER_LINK - 1] || '';
        status = String(row[CONFIG.COLS.STATUS - 1] || 'SUCCESS');
        dateProcessed = String(row[CONFIG.COLS.DATE_PROCESSED - 1] || '');
        originalName = String(row[CONFIG.COLS.ORIGINAL_NAME - 1] || '');
        fileId = extractDriveFileId(String(row[CONFIG.COLS.FILE_ID - 1] || '')) || extractDriveFileId(fileLinkFormula);

function getNextSerialNumber(sheet) {

/**
 * Rebuilds the Monthly Summary tab.
 * OPTIMIZED: one batched getFormulas() read instead of a getFormula() call per row.
 */
function refreshMonthlySummary() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const masterSheet = ss.getSheetByName(CONFIG.SHEET_MASTER_DATA);
  const summarySheet = ss.getSheetByName(CONFIG.SHEET_MONTHLY_SUMMARY) || setupMonthlySummarySheet(ss);
  if (!masterSheet) return;

  const lastRow = masterSheet.getLastRow();
  if (lastRow <= 1) {
    const sLastRow = summarySheet.getLastRow();
    if (sLastRow > 1) summarySheet.getRange(2, 1, sLastRow - 1, 3).clearContent().clearFormat();
    return;
  }

  const numRows = lastRow - 1;
  const data = masterSheet.getRange(2, 1, numRows, CONFIG.COLS.NOTES).getValues();
  const folderFormulas = masterSheet.getRange(2, CONFIG.COLS.FOLDER_LINK, numRows, 1).getFormulas();

  const monthNames = ['January', 'February', 'March', 'April', 'May', 'June',
    'July', 'August', 'September', 'October', 'November', 'December'];
  const monthMap = new Map();

  for (let i = 0; i < numRows; i++) {
    const row = data[i];
    const status = String(row[CONFIG.COLS.STATUS - 1] || '');
    let month = row[CONFIG.COLS.MONTH - 1];

    if (month instanceof Date) {
      month = monthNames[month.getMonth()] + ' ' + month.getFullYear();
    } else {
      month = String(month || '').trim();
    }

    if (status !== 'SUCCESS' || !month || month === 'UNKNOWN' || month === 'ERROR') continue;

    let folderUrl = '';
    const match = String(folderFormulas[i][0] || '').match(/HYPERLINK\("([^"]+)"/i);
    if (match) folderUrl = match[1];

    if (!monthMap.has(month)) monthMap.set(month, { count: 0, folderUrl: folderUrl });
    const entry = monthMap.get(month);
    entry.count += 1;
    if (!entry.folderUrl && folderUrl) entry.folderUrl = folderUrl;
  }

  const sLastRow = summarySheet.getLastRow();
  if (sLastRow > 1) summarySheet.getRange(2, 1, sLastRow - 1, 3).clear();
  if (monthMap.size === 0) return;

  const sortedMonths = Array.from(monthMap.keys()).sort(function (a, b) {
    const dA = new Date(Date.parse('01 ' + String(a || '')));
    const dB = new Date(Date.parse('01 ' + String(b || '')));
    if (!isNaN(dA.getTime()) && !isNaN(dB.getTime())) return dA.getTime() - dB.getTime();
    return String(a).localeCompare(String(b));
  });

  const summaryRows = sortedMonths.map(function (m) {
    const info = monthMap.get(m);
    return [m, info.count, info.folderUrl ? ('=HYPERLINK("' + info.folderUrl + '", "\uD83D\uDCC1 Open Folder")') : '\u2014'];
  });

  const targetRange = summarySheet.getRange(2, 1, summaryRows.length, 3);
  targetRange.setValues(summaryRows);
  targetRange.setFontFamily('Roboto').setFontSize(10).setVerticalAlignment('middle');
  summarySheet.getRange(2, 1, summaryRows.length, 1).setFontWeight('bold');
  summarySheet.getRange(2, 2, summaryRows.length, 2).setHorizontalAlignment('center');
  summarySheet.setRowHeights(2, summaryRows.length, 26);
}

function testGeminiConnection() {

                '/models/' + testModel + ':generateContent?key=' + apiKey;
    try {
      const res = UrlFetchApp.fetch(
        'https://generativelanguage.googleapis.com/' + CONFIG.GEMINI_API_VERSION +
        '/models/' + testModel + ':generateContent',

        contentType: 'application/json',

      const code = res.getResponseCode();
      lastBody = res.getContentText();
      notes.push(testModel + ' -> HTTP ' + code);
      if (code === 200) {
        _ACTIVE_WORKING_MODEL = testModel;
        try { props.setProperty(CONFIG.PROP_ACTIVE_MODEL, testModel); } catch (e) {}
        let extra = '';
        if (testModel !== CONFIG.GEMINI_MODEL) {
          extra = '\n\nPreferred ' + CONFIG.GEMINI_MODEL + ' returned 404 for this API key ' +
                  '(Google now blocks 2.5 Flash-Lite for many new keys). ' +
                  'Using ' + testModel + ' instead — same Flash-Lite class, fastest available.';
        }
        ui.alert('Gemini Connection: SUCCESS',
          'Connected to the Google Gemini API.\n\n' +
          '\u2022 Active Model: ' + testModel + extra + '\n\n' +
          notes.join('\n'),

    '\n\n' + String(lastBody).substring(0, 400),
    ui.ButtonSet.OK);
}

/**
 * Full extraction diagnostic. Shows exactly why a PDF fails: which model was used,
 * whether text extraction produced real text or stream noise, the HTTP status, and
 * the raw API reply.
 */
function diagnoseOneFile() {
  const props = PropertiesService.getScriptProperties();
  const apiKey = props.getProperty(CONFIG.PROP_GEMINI_KEY);
  const reviewId = props.getProperty(CONFIG.PROP_NEEDS_REVIEW_FOLDER_ID);
  const inboxId = props.getProperty(CONFIG.PROP_INBOX_FOLDER_ID);
  const out = [];

  out.push('API KEY: ' + (apiKey ? 'present' : 'MISSING'));

  let file = null;
  [reviewId, inboxId].forEach(function (id) {
    if (file || !id) return;
    try {
      const folder = DriveApp.getFolderById(id);
      out.push('SCANNING: ' + folder.getName());
      const it = folder.getFiles();
      while (it.hasNext()) {
        const f = it.next();
        if (!file && f.getName().toLowerCase().indexOf('.pdf') !== -1) file = f;
      }
    } catch (e) {
      out.push('FOLDER ERROR: ' + e.message);
    }
  });

  if (!file) {
    out.push('>>> NO PDF FOUND in NEEDS REVIEW or INBOX.');
    Logger.log(out.join('\n'));
    try { SpreadsheetApp.getUi().alert(out.join('\n')); } catch (e) {}
    return;
  }

  out.push('FILE: ' + file.getName());
  out.push('SIZE: ' + file.getSize() + ' bytes');

  try {
    const blob = file.getBlob();
    const txt = extractTextFromPdfBlob(blob);
    if (txt) {
      out.push('MODE: TEXT (' + txt.length + ' chars) - ' + scoreExtractedText(txt).why);
      out.push('SAMPLE: ' + txt.substring(0, 200));
    } else {
      out.push('MODE: FULL PDF (no usable text layer - correct for scanned or compressed PDFs)');
    }

    const model = resolveWorkingGeminiModel(apiKey);
    out.push('MODEL: ' + model + ' (Flash-Lite, thinking off/minimal)');

    const built = buildGeminiRequest(blob, apiKey, model, true);
    const r = UrlFetchApp.fetch(built.req.url, built.req);
    const code = r.getResponseCode();
    out.push('HTTP: ' + code);

    if (code === 503) {
      out.push('>>> The model is temporarily overloaded on Google\'s side. This is not a');
      out.push('    problem with your file or key. Wait 2-3 minutes and try again.');
    } else if (code === 429) {
      out.push('>>> Rate limit reached. Wait a minute and try again.');
    }

    const parsed = parseGeminiResponse(r);
    out.push('PARSE RESULT: ' + (parsed.data ? 'OK' : parsed.reason));
    if (parsed.data) out.push('EXTRACTED: ' + JSON.stringify(parsed.data));
    out.push('RAW REPLY: ' + r.getContentText().substring(0, 900));
  } catch (err) {
    out.push('EXCEPTION: ' + err.message);
  }

  Logger.log(out.join('\n'));
  try { SpreadsheetApp.getUi().alert(out.join('\n').substring(0, 1400)); } catch (e) {}
}

/**
 * Dashboard payload.
 * OPTIMIZED: no Drive sharing call per row, so dashboard loads are near-instant
 * regardless of how many clients are in the sheet.
 */
function fetchDashboardData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) return { error: 'Spreadsheet not accessible.' };

  const masterSheet = ss.getSheetByName(CONFIG.SHEET_MASTER_DATA);
  if (!masterSheet) return { error: 'Master Data sheet not found. Please run system setup.' };

  ensureMasterDataSchema(masterSheet);

  const lastRow = masterSheet.getLastRow();
  if (lastRow <= 1) {
    return {
      stats: { total: 0, dueIn7Days: 0, dueIn30Days: 0, overdue: 0, needsReview: 0 },
      months: [], companies: [], clients: []
    };
  }

  const numRows = lastRow - 1;
  const data = masterSheet.getRange(2, 1, numRows, CONFIG.COLS.NOTES).getValues();
  const formulas = masterSheet.getRange(2, 1, numRows, CONFIG.COLS.NOTES).getFormulas();

  const clients = [];
  const monthSet = new Set();
  const companySet = new Set();
  let dueIn7Days = 0, dueIn30Days = 0, overdue = 0, needsReview = 0;

  const today = new Date();
  today.setHours(0, 0, 0, 0);

  for (let i = 0; i < numRows; i++) {
    const row = data[i];
    const formRow = formulas[i];

    const clientName = String(row[CONFIG.COLS.CLIENT_NAME - 1] || '').trim();
    const renewalDate = String(row[CONFIG.COLS.RENEWAL_DATE - 1] || '').trim();
    const month = String(row[CONFIG.COLS.MONTH - 1] || '').trim();
    const company = String(row[CONFIG.COLS.COMPANY - 1] || '').trim();
    const policyNo = String(row[CONFIG.COLS.POLICY_NO - 1] || '').trim();
    const mobile = String(row[CONFIG.COLS.CLIENT_MOBILE - 1] || '').trim();
    const email = String(row[CONFIG.COLS.CLIENT_EMAIL - 1] || '').trim();
    const status = String(row[CONFIG.COLS.STATUS - 1] || 'SUCCESS').trim();
    const notes = String(row[CONFIG.COLS.NOTES - 1] || '').trim();

    const fileFormula = String(formRow[CONFIG.COLS.FILE_LINK - 1] || '');
    const folderFormula = String(formRow[CONFIG.COLS.FOLDER_LINK - 1] || '');
    const fileId = extractDriveFileId(String(row[CONFIG.COLS.FILE_ID - 1] || '').trim()) || extractDriveFileId(fileFormula);
    const fileUrl = buildDriveViewUrl(fileId);

    let folderUrl = '';
    const folderMatch = folderFormula.match(/HYPERLINK\("([^"]+)"/i);
    if (folderMatch) folderUrl = folderMatch[1];

    if (month && month !== 'UNKNOWN' && month !== 'ERROR') monthSet.add(month);
    if (company && company !== 'UNKNOWN' && company !== 'ERROR') companySet.add(company);

    let daysRemaining = null;
    if (renewalDate && /^\d{4}-\d{2}-\d{2}$/.test(renewalDate)) {
      const renDate = new Date(renewalDate);
      if (!isNaN(renDate.getTime())) {
        renDate.setHours(0, 0, 0, 0);
        daysRemaining = Math.round((renDate - today) / (1000 * 60 * 60 * 24));
        if (daysRemaining < 0) overdue++;
        else if (daysRemaining <= 7) dueIn7Days++;
        else if (daysRemaining <= 30) dueIn30Days++;
      }
    }

    if (status === 'NEEDS REVIEW' || status === 'ERROR') needsReview++;

    clients.push({
      rowIndex: i + 2, srNo: row[CONFIG.COLS.SR_NO - 1],
      clientName: clientName, renewalDate: renewalDate, month: month, company: company,
      policyNo: policyNo, mobile: mobile, email: email, status: status,
      fileUrl: fileUrl, folderUrl: folderUrl,
      whatsappUrl: generateWhatsAppLink(mobile, clientName, company, renewalDate, fileUrl, policyNo),
      gmailUrl: generateGmailComposeLink(email, clientName, company, renewalDate, fileUrl, policyNo),
      daysRemaining: daysRemaining, notes: notes
    });
  }

  clients.sort(function (a, b) {
    if (!a.renewalDate) return 1;
    if (!b.renewalDate) return -1;
    return a.renewalDate.localeCompare(b.renewalDate);
  });

  return {
    stats: {
      total: clients.length, dueIn7Days: dueIn7Days, dueIn30Days: dueIn30Days,
      overdue: overdue, needsReview: needsReview
    },
    months: Array.from(monthSet),
    companies: Array.from(companySet).sort(),
    clients: clients
  };
}

/**
 * Uploads base64 PDFs to INBOX and runs the pipeline.
 * OPTIMIZED: no per-file setSharing on upload; sharing happens in the parallel batch.
 */
function uploadFilesAndProcess(filesList) {
  const props = PropertiesService.getScriptProperties();
  const inboxId = props.getProperty(CONFIG.PROP_INBOX_FOLDER_ID);

  if (!inboxId || !folderExists(inboxId)) {
    throw new Error('Inbox folder not found. Please run System Setup first.');
  }

  const inboxFolder = DriveApp.getFolderById(inboxId);
  let uploadCount = 0;

  if (Array.isArray(filesList)) {
    filesList.forEach(function (fileObj) {
      if (fileObj.base64 && fileObj.name) {
        const decoded = Utilities.base64Decode(fileObj.base64);
        inboxFolder.createFile(Utilities.newBlob(decoded, fileObj.type || 'application/pdf', fileObj.name));
        uploadCount++;
      }
    });
  }

  const stats = processInbox();
  return {
    uploaded: uploadCount, processed: stats.processed, success: stats.success,
    needsReview: stats.needsReview, skipped: stats.skipped, remaining: stats.remaining,
    overloaded: stats.overloaded, elapsedSeconds: stats.elapsedSeconds
  };
}

<script>
var allClients = [];
var queuedFiles = [];

window.onload = function () { loadDashboard(); setupDragDrop(); };

function loadDashboard() {
  document.getElementById('loading-state').classList.remove('hidden');
  document.getElementById('client-grid').classList.add('hidden');
  document.getElementById('empty-state').classList.add('hidden');
  google.script.run.withSuccessHandler(renderDashboard).withFailureHandler(handleError).fetchDashboardData();
}

function renderDashboard(data) {
  document.getElementById('loading-state').classList.add('hidden');
  if (data.error) { alert('Error: ' + data.error); return; }
  document.getElementById('kpi-total').innerText = data.stats.total;
  document.getElementById('kpi-urgent').innerText = data.stats.dueIn7Days;
  document.getElementById('kpi-month').innerText = data.stats.dueIn30Days;
  document.getElementById('kpi-review').innerText = data.stats.needsReview;

  var ms = document.getElementById('filter-month');
  ms.innerHTML = '<option value="ALL">All Months</option>';
  data.months.forEach(function (m) {
    var o = document.createElement('option'); o.value = m; o.innerText = m; ms.appendChild(o);
  });

  var cs = document.getElementById('filter-company');
  cs.innerHTML = '<option value="ALL">All Insurers</option>';
  data.companies.forEach(function (c) {
    var o = document.createElement('option'); o.value = c; o.innerText = c; cs.appendChild(o);
  });

  allClients = data.clients || [];
  filterClients();
}

function filterClients() {
  var q = document.getElementById('search-input').value.toLowerCase().trim();
  var sm = document.getElementById('filter-month').value;
  var sc = document.getElementById('filter-company').value;
  var su = document.getElementById('filter-urgency').value;

  var filtered = allClients.filter(function (c) {
    var mq = !q || c.clientName.toLowerCase().indexOf(q) !== -1 ||
      c.policyNo.toLowerCase().indexOf(q) !== -1 || c.mobile.toLowerCase().indexOf(q) !== -1 ||
      c.email.toLowerCase().indexOf(q) !== -1 || c.company.toLowerCase().indexOf(q) !== -1;
    var mm = (sm === 'ALL') || (c.month === sm);
    var mc = (sc === 'ALL') || (c.company === sc);
    var mu = true;
    if (su === 'URGENT') mu = c.daysRemaining !== null && c.daysRemaining >= 0 && c.daysRemaining <= 7;
    else if (su === 'MONTH') mu = c.daysRemaining !== null && c.daysRemaining >= 0 && c.daysRemaining <= 30;
    else if (su === 'OVERDUE') mu = c.daysRemaining !== null && c.daysRemaining < 0;
    else if (su === 'REVIEW') mu = c.status === 'NEEDS REVIEW' || c.status === 'ERROR';
    return mq && mm && mc && mu;
  });

  renderClientCards(filtered);
}

function renderClientCards(clients) {
  var grid = document.getElementById('client-grid');
  var empty = document.getElementById('empty-state');

  if (!clients || clients.length === 0) {
    grid.innerHTML = ''; grid.classList.add('hidden'); empty.classList.remove('hidden'); return;
  }
  empty.classList.add('hidden'); grid.classList.remove('hidden'); grid.innerHTML = '';

  clients.forEach(function (c) {
    var card = document.createElement('div');
    card.className = 'client-card';

    var tag = '';
    if (c.daysRemaining !== null) {
      if (c.daysRemaining < 0) tag = '<span class="due-tag urgent">Overdue (' + Math.abs(c.daysRemaining) + 'd ago)</span>';
      else if (c.daysRemaining <= 7) tag = '<span class="due-tag urgent">Due in ' + c.daysRemaining + ' days</span>';
      else if (c.daysRemaining <= 30) tag = '<span class="due-tag warning">' + c.daysRemaining + ' days left</span>';
      else tag = '<span class="due-tag good">' + c.daysRemaining + ' days left</span>';
    } else {
      tag = '<span class="due-tag warning">Check Date</span>';
    }

    card.innerHTML =
      '<button class="btn-edit-float" onclick="openEditModal(' + c.rowIndex + ')">Edit</button>' +
      '<div><div class="card-top"><div>' +
      '<div class="c-name">' + (c.clientName || 'Unknown Client') + '</div>' +
      '<div class="c-policy">Policy: ' + (c.policyNo || '-') + '</div></div>' +
      '<div class="badge-company">' + (c.company || 'INSURANCE') + '</div></div>' +
      '<div class="due-row"><span>Due: <b>' + (c.renewalDate || '-') + '</b></span>' + tag + '</div>' +
      '<div class="contact-details"><div>Phone: <b>' + (c.mobile || 'No phone') + '</b></div>' +
      '<div>Email: <b>' + (c.email || 'No email') + '</b></div></div></div>' +
      '<div class="card-actions">' +
      '<a href="' + c.whatsappUrl + '" target="_blank" class="btn-action btn-wa">WhatsApp</a>' +
      '<a href="' + c.gmailUrl + '" target="_blank" class="btn-action btn-gm">Gmail</a>' +
      (c.fileUrl ? '<a href="' + c.fileUrl + '" target="_blank" class="btn-action btn-pdf">View PDF Notice</a>' : '') +
      '</div>';

    grid.appendChild(card);
  });
}

function openEditModal(rowIndex) {
  var client = null;
  for (var i = 0; i < allClients.length; i++) {
    if (allClients[i].rowIndex === rowIndex) { client = allClients[i]; break; }
  }
  if (!client) return;
  document.getElementById('edit-row').value = client.rowIndex;
  document.getElementById('edit-name').value = client.clientName;
  document.getElementById('edit-policy').value = client.policyNo;
  document.getElementById('edit-mobile').value = client.mobile;
  document.getElementById('edit-email').value = client.email;
  document.getElementById('edit-modal').classList.remove('hidden');
}

function closeModal() { document.getElementById('edit-modal').classList.add('hidden'); }

function saveModalChanges() {
  var row = parseInt(document.getElementById('edit-row').value, 10);
  var p = document.getElementById('edit-policy').value.trim();
  var m = document.getElementById('edit-mobile').value.trim();
  var e = document.getElementById('edit-email').value.trim();
  closeModal();
  document.getElementById('loading-state').classList.remove('hidden');
  google.script.run.withSuccessHandler(function () { loadDashboard(); })
    .withFailureHandler(handleError).saveClientFromWeb(row, m, e, p);
}

function setupDragDrop() {
  var dz = document.getElementById('drop-zone');
  var fi = document.getElementById('file-input');
  dz.addEventListener('dragover', function (e) { e.preventDefault(); dz.classList.add('dragover'); });
  dz.addEventListener('dragleave', function () { dz.classList.remove('dragover'); });
  dz.addEventListener('drop', function (e) {
    e.preventDefault(); dz.classList.remove('dragover');
    if (e.dataTransfer.files && e.dataTransfer.files.length > 0) handleSelectedFiles(e.dataTransfer.files);
  });
  fi.addEventListener('change', function (e) {
    if (e.target.files && e.target.files.length > 0) handleSelectedFiles(e.target.files);
  });
}

function handleSelectedFiles(files) {
  for (var i = 0; i < files.length; i++) {
    var f = files[i];
    if (f.name.toLowerCase().indexOf('.pdf') !== -1 || f.type === 'application/pdf') queuedFiles.push(f);
  }
  renderQueue();
}

function renderQueue() {
  var qList = document.getElementById('queued-list');
  var qc = document.getElementById('queue-container');
  if (queuedFiles.length === 0) { qc.classList.add('hidden'); return; }
  qc.classList.remove('hidden');
  qList.innerHTML = '';
  queuedFiles.forEach(function (file, idx) {
    var item = document.createElement('div');
    item.className = 'queued-item';
    item.innerHTML = '<span>' + file.name + ' (' + Math.round(file.size / 1024) + ' KB)</span>' +
      '<span class="queued-remove" onclick="removeQueued(' + idx + ')">&#10006;</span>';
    qList.appendChild(item);
  });
}

function removeQueued(idx) { queuedFiles.splice(idx, 1); renderQueue(); }

function uploadAndProcessFiles() {
  if (queuedFiles.length === 0) return;
  var statusEl = document.getElementById('upload-status');
  var btn = document.getElementById('btn-process');
  btn.disabled = true;
  btn.innerText = 'Uploading & Processing...';
  statusEl.classList.remove('hidden');
  statusEl.innerText = 'Encoding ' + queuedFiles.length + ' PDF files...';

  var fileDataList = [];
  var pending = queuedFiles.length;

  queuedFiles.forEach(function (file) {
    var reader = new FileReader();
    reader.onload = function () {
      var b64 = reader.result.split(',')[1] || reader.result;
      fileDataList.push({ name: file.name, type: file.type || 'application/pdf', base64: b64 });
      pending--;
      if (pending === 0) sendToServer(fileDataList, statusEl, btn);
    };
    reader.onerror = function () { pending--; if (pending === 0) sendToServer(fileDataList, statusEl, btn); };
    reader.readAsDataURL(file);
  });
}

function sendToServer(fileDataList, statusEl, btn) {
  var elapsed = 0;
  statusEl.innerText = 'Processing ' + fileDataList.length + ' PDFs with Gemini AI... (0.0s)';
  var timer = setInterval(function () {
    elapsed += 0.5;
    statusEl.innerText = 'Processing ' + fileDataList.length + ' PDFs with Gemini AI... (' + elapsed.toFixed(1) + 's)';
  }, 500);

  google.script.run
    .withSuccessHandler(function (res) {
      clearInterval(timer);
      var t = res.elapsedSeconds || elapsed.toFixed(1);
      var msg = 'Done in ' + t + 's. Processed ' + res.processed + ' notices (' +
        res.success + ' successful, ' + res.needsReview + ' needs review)';
      if (res.overloaded > 0) msg += '. ' + res.overloaded + ' hit a temporary model overload - retry in a few minutes.';
      statusEl.innerText = msg;
      queuedFiles = [];
      renderQueue();
      btn.disabled = false;
      btn.innerText = 'Upload & Process with Gemini AI';
      setTimeout(function () { statusEl.classList.add('hidden'); loadDashboard(); }, 3000);
    })
    .withFailureHandler(function (err) {
      clearInterval(timer);
      handleError(err);
      btn.disabled = false;
      btn.innerText = 'Upload & Process with Gemini AI';
      statusEl.classList.add('hidden');
    })
    .uploadFilesAndProcess(fileDataList);
}

function handleError(err) {
  document.getElementById('loading-state').classList.add('hidden');
  alert('Error: ' + (err.message || err.toString()));
}
